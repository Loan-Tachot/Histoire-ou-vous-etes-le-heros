CREATE TABLE Classe(
   id INT,
   Nom_classe VARCHAR(50) NOT NULL,
   Element_ VARCHAR(50),
   Arme VARCHAR(50) NOT NULL,
   PRIMARY KEY(id)
);

CREATE TABLE Objet(
   id INT,
   Label VARCHAR(50) NOT NULL,
   Effet VARCHAR(50),
   PRIMARY KEY(id)
);

CREATE TABLE histoire(
   Id_histoire COUNTER,
   text VARCHAR(1000),
   PRIMARY KEY(Id_histoire)
);

CREATE TABLE PNJ(
   Id_PNJ COUNTER,
   Etat_actuel VARCHAR(100) NOT NULL,
   id INT NOT NULL,
   PRIMARY KEY(Id_PNJ),
   FOREIGN KEY(id) REFERENCES Classe(id)
);

CREATE TABLE personnage(
   Id INT,
   Nom VARCHAR(50) NOT NULL,
   PV INT NOT NULL,
   Force INT NOT NULL,
   Agilité INT NOT NULL,
   PM INT NOT NULL,
   Puissance INT NOT NULL,
   argent INT NOT NULL,
   Compteur_de_dodo COUNTER NOT NULL,
   id_1 INT NOT NULL,
   PRIMARY KEY(Id),
   UNIQUE(id_1),
   FOREIGN KEY(id_1) REFERENCES Classe(id)
);

CREATE TABLE Inventaire(
   Id INT,
   id_1 INT,
   Nb_objet INT NOT NULL,
   PRIMARY KEY(Id, id_1),
   FOREIGN KEY(Id) REFERENCES personnage(Id),
   FOREIGN KEY(id_1) REFERENCES Objet(id)
);

CREATE TABLE déboucher(
   Id_histoire INT,
   Id_histoire_1 INT,
   Text VARCHAR(50),
   PRIMARY KEY(Id_histoire, Id_histoire_1),
   FOREIGN KEY(Id_histoire) REFERENCES histoire(Id_histoire),
   FOREIGN KEY(Id_histoire_1) REFERENCES histoire(Id_histoire)
);

CREATE TABLE Relation(
   Id INT,
   Id_PNJ INT,
   relation_avec_pp VARCHAR(50),
   PRIMARY KEY(Id, Id_PNJ),
   FOREIGN KEY(Id) REFERENCES personnage(Id),
   FOREIGN KEY(Id_PNJ) REFERENCES PNJ(Id_PNJ)
);
