CREATE TABLE Classe(
   id INT AUTO_INCREMENT,
   Nom_classe VARCHAR(50) NOT NULL,
   Element_ VARCHAR(50),
   Arme VARCHAR(50) NOT NULL,
   PRIMARY KEY(id)
) ENGINE=InnoDB;

CREATE TABLE Objet(
   id INT AUTO_INCREMENT,
   Label VARCHAR(50) NOT NULL,
   Effet VARCHAR(50),
   PRIMARY KEY(id)
) ENGINE=InnoDB;

CREATE TABLE Amelioration(
   Id_Amelioration INT AUTO_INCREMENT,
   bonus_PV INT,
   bonus_Force INT,
   bonus_Agi INT,
   Bonus_PM INT,
   bonus_Puissance INT,
   Bonus_Argent INT,
   PRIMARY KEY(Id_Amelioration)
) ENGINE=InnoDB;

CREATE TABLE personnage(
   Id INT AUTO_INCREMENT,
   PV_base INT NOT NULL,
   Force_base INT NOT NULL,
   Agilite_base INT NOT NULL,
   PM_base INT NOT NULL,
   Puissance_base INT NOT NULL,
   argent_base INT NOT NULL,
   id_1 INT NOT NULL,
   PRIMARY KEY(Id),
   UNIQUE(id_1),
   FOREIGN KEY(id_1) REFERENCES Classe(id)
) ENGINE=InnoDB;

CREATE TABLE histoire(
   Id_histoire INT AUTO_INCREMENT,
   texte_histoire VARCHAR(1000),
   Id_Amelioration INT NOT NULL,
   PRIMARY KEY(Id_histoire),
   FOREIGN KEY(Id_Amelioration) REFERENCES Amelioration(Id_Amelioration)
) ENGINE=InnoDB;

CREATE TABLE Deboucher(
   Id_Deboucher INT AUTO_INCREMENT,
   choix VARCHAR(50),
   Id_histoire INT NOT NULL,
   PRIMARY KEY(Id_Deboucher),
   FOREIGN KEY(Id_histoire) REFERENCES histoire(Id_histoire)
) ENGINE=InnoDB;

CREATE TABLE obtenue(
   id INT,
   Id_histoire INT,
   PRIMARY KEY(id, Id_histoire),
   FOREIGN KEY(id) REFERENCES Objet(id),
   FOREIGN KEY(Id_histoire) REFERENCES histoire(Id_histoire)
) ENGINE=InnoDB;
