use tachot;

SET FOREIGN_KEY_CHECKS = 0;

DROP TABLE IF EXISTS 
    Deboucher,
    obtenue,
    histoire,
    Objet,
    Classe,
    Amelioration;

SET FOREIGN_KEY_CHECKS = 1;
-- =========================================================
-- TABLES
-- =========================================================

CREATE TABLE Amelioration(
   Id_Amelioration INT AUTO_INCREMENT,
   bonus_PV        INT DEFAULT 0,
   bonus_Force     INT DEFAULT 0,
   bonus_Agi       INT DEFAULT 0,
   Bonus_PM        INT DEFAULT 0,
   bonus_Puissance INT DEFAULT 0,
   Bonus_Argent    INT DEFAULT 0,
   PRIMARY KEY(Id_Amelioration)
) ENGINE=InnoDB;

CREATE TABLE histoire(
   Id_histoire    INT AUTO_INCREMENT,
   texte_histoire VARCHAR(1000) NOT NULL,
   Id_Amelioration INT NOT NULL,
   PRIMARY KEY(Id_histoire),
   FOREIGN KEY(Id_Amelioration) REFERENCES Amelioration(Id_Amelioration)
) ENGINE=InnoDB;

-- condition_requise : expression PHP évaluée côté serveur
-- ex: Force>=8  PM<8  Agi>4  PM>Force  Force>PM  Force==PM
CREATE TABLE Deboucher(
   Id_Deboucher          INT AUTO_INCREMENT,
   choix                 VARCHAR(100) NOT NULL,
   condition_requise     VARCHAR(100) DEFAULT NULL,
   Id_histoire_source    INT NOT NULL,
   Id_histoire_destination INT NOT NULL,
   PRIMARY KEY(Id_Deboucher),
   FOREIGN KEY(Id_histoire_source)      REFERENCES histoire(Id_histoire),
   FOREIGN KEY(Id_histoire_destination) REFERENCES histoire(Id_histoire)
) ENGINE=InnoDB;

CREATE TABLE Classe(
   id         INT AUTO_INCREMENT,
   Nom_classe VARCHAR(50) NOT NULL,
   Element_   VARCHAR(50),
   Arme       VARCHAR(50) NOT NULL,
   PRIMARY KEY(id)
) ENGINE=InnoDB;

CREATE TABLE Objet(
   id    INT AUTO_INCREMENT,
   Label VARCHAR(100) NOT NULL,
   Effet VARCHAR(255),
   PRIMARY KEY(id)
) ENGINE=InnoDB;

CREATE TABLE obtenue(
   id_objet    INT NOT NULL,
   Id_histoire INT NOT NULL,
   PRIMARY KEY(id_objet, Id_histoire),
   FOREIGN KEY(id_objet)    REFERENCES Objet(id),
   FOREIGN KEY(Id_histoire) REFERENCES histoire(Id_histoire)
) ENGINE=InnoDB;
